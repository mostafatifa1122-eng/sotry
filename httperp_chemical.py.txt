# =========================== نماذج المستخدمين والصلاحيات ===========================

@dataclass
class User:
    id: int
    username: str
    password_hash: str  # مش هنخزن كلمة المرور نصاً
    full_name: str
    role: str  # manager, assistant_manager, employee, lab_tech, warehouse
    employee_id: Optional[int] = None  # لو الموظف مرتبط بموظف في نظام الـ HR
    is_active: bool = True
    created_at: datetime = None
    last_login: Optional[datetime] = None

@dataclass
class Permission:
    id: int
    role: str
    module: str  # products, invoices, batches, lab, attendance, payroll, etc.
    can_view: bool = False
    can_create: bool = False
    can_edit: bool = False
    can_delete: bool = False

# =========================== إضافة جداول المستخدمين في قاعدة البيانات ===========================

def create_user_tables(self):
    tables = [
        """CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            full_name TEXT NOT NULL,
            role TEXT NOT NULL,
            employee_id INTEGER,
            is_active INTEGER DEFAULT 1,
            created_at TEXT NOT NULL,
            last_login TEXT,
            FOREIGN KEY (employee_id) REFERENCES employees(id)
        )""",
        
        """CREATE TABLE IF NOT EXISTS permissions (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            role TEXT NOT NULL,
            module TEXT NOT NULL,
            can_view INTEGER DEFAULT 0,
            can_create INTEGER DEFAULT 0,
            can_edit INTEGER DEFAULT 0,
            can_delete INTEGER DEFAULT 0
        )"""
    ]
    for table in tables:
        try:
            self.cursor.execute(table)
        except:
            pass
    self.conn.commit()

# =========================== دوال إدارة المستخدمين ===========================

class UserManager:
    def __init__(self, db):
        self.db = db

    def hash_password(self, password):
        return hashlib.sha256(password.encode()).hexdigest()

    def create_user(self, username, password, full_name, role, employee_id=None):
        """ إنشاء مستخدم جديد """
        password_hash = self.hash_password(password)
        created_at = datetime.now().isoformat()
        try:
            self.db.execute(
                """INSERT INTO users 
                   (username, password_hash, full_name, role, employee_id, is_active, created_at, last_login)
                   VALUES (?, ?, ?, ?, ?, 1, ?, NULL)""",
                (username, password_hash, full_name, role, employee_id, created_at)
            )
            return True, "تم إنشاء المستخدم بنجاح"
        except sqlite3.IntegrityError:
            return False, "اسم المستخدم موجود بالفعل"

    def login(self, username, password):
        """ تسجيل الدخول """
        password_hash = self.hash_password(password)
        row = self.db.fetch_one(
            "SELECT * FROM users WHERE username = ? AND password_hash = ? AND is_active = 1",
            (username, password_hash)
        )
        if row:
            # تحديث آخر تسجيل دخول
            self.db.execute(
                "UPDATE users SET last_login = ? WHERE id = ?",
                (datetime.now().isoformat(), row[0])
            )
            user = dict(zip(['id','username','password_hash','full_name','role','employee_id','is_active','created_at','last_login'], row))
            return True, user
        return False, "اسم المستخدم أو كلمة المرور غير صحيحة"

    def get_user_permissions(self, role):
        """ جلب صلاحيات المستخدم حسب دوره """
        rows = self.db.fetch_all(
            "SELECT * FROM permissions WHERE role = ?",
            (role,)
        )
        permissions = {}
        for row in rows:
            perm = dict(zip(['id','role','module','can_view','can_create','can_edit','can_delete'], row))
            permissions[perm['module']] = perm
        return permissions

    def check_permission(self, role, module, action):
        """ التحقق من صلاحية معينة """
        row = self.db.fetch_one(
            "SELECT * FROM permissions WHERE role = ? AND module = ?",
            (role, module)
        )
        if not row:
            # إذا لم توجد صلاحيات محددة، نستخدم الـ default حسب الدور
            return self.get_default_permission(role, module, action)
        
        perm = dict(zip(['id','role','module','can_view','can_create','can_edit','can_delete'], row))
        if action == 'view':
            return bool(perm['can_view'])
        elif action == 'create':
            return bool(perm['can_create'])
        elif action == 'edit':
            return bool(perm['can_edit'])
        elif action == 'delete':
            return bool(perm['can_delete'])
        return False

    def get_default_permission(self, role, module, action):
        """ صلاحيات افتراضية لكل دور """
        # المدير العام: كل حاجة
        if role == 'manager':
            return True
        
        # مساعد المدير: كل حاجة ما عدا حذف
        if role == 'assistant_manager':
            if action == 'delete':
                return False
            return True
        
        # الموظف: بس يشوف معلوماته
        if role == 'employee':
            if module in ['attendance', 'payroll'] and action == 'view':
                return True
            return False
        
        # مسؤول المعمل: المعمل والجودة فقط
        if role == 'lab_tech':
            if module in ['lab', 'quality_control']:
                return action != 'delete'
            return False
        
        # مسؤول المخازن: المنتجات والفواتير فقط
        if role == 'warehouse':
            if module in ['products', 'invoices', 'batches']:
                return action != 'delete'
            return False
        
        return False

    def assign_permission(self, role, module, can_view, can_create, can_edit, can_delete):
        """ تعيين صلاحيات لدور معين """
        existing = self.db.fetch_one(
            "SELECT id FROM permissions WHERE role = ? AND module = ?",
            (role, module)
        )
        if existing:
            self.db.execute(
                """UPDATE permissions 
                   SET can_view = ?, can_create = ?, can_edit = ?, can_delete = ? 
                   WHERE role = ? AND module = ?""",
                (1 if can_view else 0, 1 if can_create else 0, 1 if can_edit else 0, 1 if can_delete else 0, role, module)
            )
        else:
            self.db.execute(
                """INSERT INTO permissions 
                   (role, module, can_view, can_create, can_edit, can_delete)
                   VALUES (?, ?, ?, ?, ?, ?)""",
                (role, module, 1 if can_view else 0, 1 if can_create else 0, 1 if can_edit else 0, 1 if can_delete else 0)
            )
        return True

    def get_all_users(self):
        rows = self.db.fetch_all("SELECT * FROM users")
        return [dict(zip(['id','username','password_hash','full_name','role','employee_id','is_active','created_at','last_login'], row)) for row in rows]

# =========================== إعداد الصلاحيات الافتراضية ===========================

def setup_default_permissions(user_manager):
    """ إعداد الصلاحيات الافتراضية لكل دور """
    modules = ['products', 'invoices', 'batches', 'lab', 'sds', 'attendance', 
               'payroll', 'employees', 'quality_control', 'maintenance', 'waste', 
               'shipping', 'complaints', 'doc', 'coa', 'supplier_audits', 'emergency_plans']
    
    # صلاحيات المدير العام: كل حاجة
    for module in modules:
        user_manager.assign_permission('manager', module, True, True, True, True)
    
    # صلاحيات مساعد المدير: كل حاجة ما عدا الحذف
    for module in modules:
        user_manager.assign_permission('assistant_manager', module, True, True, True, False)
    
    # صلاحيات الموظف: بس الحضور والرواتب (عرض فقط)
    user_manager.assign_permission('employee', 'attendance', True, False, False, False)
    user_manager.assign_permission('employee', 'payroll', True, False, False, False)
    
    # صلاحيات مسؤول المعمل
    user_manager.assign_permission('lab_tech', 'lab', True, True, True, False)
    user_manager.assign_permission('lab_tech', 'quality_control', True, True, True, False)
    user_manager.assign_permission('lab_tech', 'sds', True, True, True, False)
    user_manager.assign_permission('lab_tech', 'coa', True, True, True, False)
    user_manager.assign_permission('lab_tech', 'batches', True, True, False, False)
    
    # صلاحيات مسؤول المخازن
    user_manager.assign_permission('warehouse', 'products', True, True, True, False)
    user_manager.assign_permission('warehouse', 'batches', True, True, True, False)
    user_manager.assign_permission('warehouse', 'invoices', True, True, True, False)
    user_manager.assign_permission('warehouse', 'shipping', True, True, True, False)

def manage_products(erp, user_manager, role):
    """ إدارة المنتجات مع التحقق من الصلاحيات """
    while True:
        print("\n📦 إدارة المنتجات والمخازن:")
        print(" 1. عرض المنتجات")
        if user_manager.check_permission(role, 'products', 'create'):
            print(" 2. إضافة منتج جديد")
        if user_manager.check_permission(role, 'products', 'edit'):
            print(" 3. تحديث المخزون")
        print(" 4. عرض المنتجات منخفضة المخزون")
        print(" 5. إذن صرف خامات")
        print(" 6. إذن استرجاع خامات")
        print(" 0. رجوع")
        
        sub = input("اختر: ")
        if sub == "0":
            break
        elif sub == "1":
            products = erp.get_products()
            for p in products:
                print(f"🔹 {p['code']} - {p['name']} | HS: {p['hs_code']} | الكمية: {p['current_quantity']} {p['unit']}")
        elif sub == "2":
            if user_manager.check_permission(role, 'products', 'create'):
                name = input("اسم المنتج: ")
                code = input("الكود: ")
                hs_code = input("رمز HS Code: ")
                category = input("النوع (raw/finished/chemical): ")
                unit = input("الوحدة (kg/liter/ton/piece): ")
                price = float(input("سعر البيع: "))
                cost = float(input("تكلفة الشراء: "))
                min_qty = float(input("الحد الأدنى للمخزون: ") or 0)
                erp.add_product(name, code, hs_code, category, unit, price, cost, min_qty)
                print("✅ تم إضافة المنتج مع رمز HS")
            else:
                print("❌ ليس لديك صلاحية لإضافة منتجات")
        elif sub == "3":
            if user_manager.check_permission(role, 'products', 'edit'):
                product_id = int(input("رقم المنتج: "))
                qty = float(input("الكمية: "))
                op = input("عملية (add/subtract): ")
                erp.update_stock(product_id, qty, op)
                print("✅ تم تحديث المخزون")
            else:
                print("❌ ليس لديك صلاحية لتحديث المخزون")
